def _parent_shas(commit: Mapping[str, Any]) -> list[str]:
    parents = commit.get("parents")
    if not isinstance(parents, list):
        raise SourceRecordRejected("SOURCE_SCHEMA_DRIFT", "commit parents missing")
    values: list[str] = []
    for index, parent in enumerate(parents):
        if not isinstance(parent, Mapping):
            raise SourceRecordRejected("SOURCE_SCHEMA_DRIFT", "commit parent malformed")
        value = parent.get("sha")
        if not isinstance(value, str) or SHA1_PATTERN.fullmatch(value) is None:
            raise SourceRecordRejected(
                "SOURCE_SCHEMA_DRIFT", f"commit parent {index} is malformed"
            )
        values.append(value)
    return values

                ["git", "-C", str(bare), "cat-file", "blob", entry.object_sha]
            )
            descriptor = context.raw_store.put(blob, media_type=SOURCE_BLOB_MEDIA_TYPE)

                "object_sha": entry.object_sha,

            }
    return {
        "repository": repository,
        "revision": revision,
        "commit_object": commit_descriptor,
        "tree_listing": tree_descriptor,
        "selected_blob_count": len(blobs),
        "blobs": blobs,

def _last_attempt_record(request: Mapping[str, Any]) -> Mapping[str, Any]:
    attempts = request.get("attempts")
    if not isinstance(attempts, list) or not attempts:
        raise DiscoveryExecutionError("RAW_CAPTURE_INCOMPLETE", "request has no attempts")
    value = attempts[-1]
    if not isinstance(value, Mapping):
        raise DiscoveryExecutionError("RAW_CAPTURE_INCOMPLETE", "attempt is malformed")
    return value


class GitHubEvidence:
    """Deduplicated API capture index with no scientific interpretation."""

    def __init__(self, context: CaptureContext | None = None) -> None:
        self.context = context
        self.entries: list[dict[str, Any]] = []
        self._cache: dict[str, tuple[Any | None, dict[str, Any]]] = {}

    @staticmethod
    def _key(
        *,
        method: str,
        endpoint: str,
        ordered_parameters: Sequence[tuple[str, str]],
        json_body: Mapping[str, Any] | None,
        capture_scope: str | None = None,
    ) -> str:
        key_material: dict[str, Any] = {
            "method": method,
            "endpoint": endpoint,
            "ordered_parameters": [list(item) for item in ordered_parameters],
            "json_body": json_body,
        }
        if capture_scope is not None:
            key_material["capture_scope"] = capture_scope
        return "CMVP-API-03-" + sha256_bytes(
            canonical_document_bytes(key_material)
        )

    def capture(
        self,
        *,
        method: str,
        endpoint: str,
        ordered_parameters: Sequence[tuple[str, str]] = (),
        json_body: Mapping[str, Any] | None = None,
        exact_query: str | None = None,
        partition_path: str | None = None,
        partition_start_utc: str | None = None,
        partition_end_utc: str | None = None,
        page: int | None = None,
        graphql_cursor: str | None = None,
        sort: str | None = None,
        order: str | None = None,
        allow_one_redirect: bool = True,
        capture_scope: str | None = None,
    ) -> tuple[Any | None, dict[str, Any]]:
        if self.context is None:
            raise DiscoveryExecutionError("CAPTURE_CONTEXT_MISSING", endpoint)
        key = self._key(
            method=method,
            endpoint=endpoint,
            ordered_parameters=ordered_parameters,
            json_body=json_body,
            capture_scope=capture_scope,
        )
        cached = self._cache.get(key)
        if cached is not None:
            return cached
        try:
            payload, request = self.context.api.request_json(
                method=method,
                endpoint=endpoint,
                ordered_parameters=ordered_parameters,
                exact_query=exact_query,
                partition_path=partition_path,
                partition_start_utc=partition_start_utc,
                partition_end_utc=partition_end_utc,
                page=page,
                graphql_cursor=graphql_cursor,
                sort=sort,
                order=order,
                json_body=json_body,
                allow_one_redirect=allow_one_redirect,
            )
            last = _last_attempt_record(request)
            entry = {
                "evidence_key": key,
                "request_number": request["request_number"],
                "status": "AVAILABLE",
                "http_status": last["http_status"],
                "raw_artifact": last["body"],
                "request_body": json_body,
            }
            result = (payload, entry)
        except DiscoveryExecutionError as error:
            if error.http_status not in {404, 410}:
                raise
            request = self.context.api.requests[-1]
            last = _last_attempt_record(request)
            entry = {
                "evidence_key": key,
                "request_number": request["request_number"],
                "status": "UNAVAILABLE",
                "http_status": error.http_status,
                "raw_artifact": last["body"],
                "request_body": json_body,
            }
            result = (None, entry)
        if capture_scope is not None:
            entry["capture_scope"] = capture_scope
        self.entries.append(entry)
        self._cache[key] = result
        return result

    def read_payload(
        self, *, root: Path, entry: Mapping[str, Any]
    ) -> Mapping[str, Any] | list[Any] | None:
        if entry.get("status") == "UNAVAILABLE":
            return None
        descriptor = entry.get("raw_artifact")
        if not isinstance(descriptor, Mapping):
            raise DiscoveryExecutionError("RAW_CAPTURE_INCOMPLETE", "missing API body")
        path = root / str(descriptor.get("path"))
        data = path.read_bytes()
        if sha256_bytes(data) != descriptor.get("sha256"):
            raise DiscoveryExecutionError(
                "RAW_ARTIFACT_HASH_MISMATCH", f"API evidence mismatch: {path}"
            )
        try:
            payload = json.loads(data)
        except (UnicodeDecodeError, json.JSONDecodeError) as error:
            raise DiscoveryExecutionError("SOURCE_SCHEMA_DRIFT", str(error)) from error
        if not isinstance(payload, (Mapping, list)):
            raise DiscoveryExecutionError("SOURCE_SCHEMA_DRIFT", "API JSON is scalar")
        return payload


def _api_repo_endpoint(owner: str, repository: str) -> str:
    return (
        "https://api.github.com/repos/"
        f"{quote(owner, safe='')}/{quote(repository, safe='')}"
    )


def _repository_coordinates_from_api_url(value: Any) -> tuple[str, str]:
    if not isinstance(value, str):
        raise DiscoveryExecutionError(
            "SOURCE_SCHEMA_DRIFT", "repository_url must be a string"
        )
    match = API_REPOSITORY_PATTERN.fullmatch(value)
    if match is None:
        raise DiscoveryExecutionError(
            "SOURCE_SCHEMA_DRIFT", "repository_url is not canonical GitHub API URL"
        )
    return match.group(1), match.group(2)


def _mapping(value: Any, label: str) -> Mapping[str, Any]:
    if not isinstance(value, Mapping):
        raise DiscoveryExecutionError("SOURCE_SCHEMA_DRIFT", f"{label} must be object")
    return value


def _positive_int(value: Any, label: str) -> int:
    if isinstance(value, bool) or not isinstance(value, int) or value <= 0:
        raise DiscoveryExecutionError(
            "SOURCE_SCHEMA_DRIFT", f"{label} must be positive integer"
        )
    return value


def _sha(value: Any, label: str) -> str:
    if not isinstance(value, str) or SHA1_PATTERN.fullmatch(value) is None:
        raise DiscoveryExecutionError(
            "SOURCE_SCHEMA_DRIFT", f"{label} must be lowercase 40-hex"
        )
    return value

def _validate_graphql_pr_node(
    node: Mapping[str, Any], *, pr_metadata: Mapping[str, Any]
) -> tuple[int, int]:
    number = node.get("number")
    repository_node = node.get("repository")
    if (
        isinstance(number, bool)
        or not isinstance(number, int)
        or number <= 0
        or not isinstance(repository_node, Mapping)
    ):
        raise SourceRecordRejected(
            "GITHUB_ISSUE_RELATION_INCOMPLETE", "merged PR node lacks identity"
        )
    repository_id = repository_node.get("databaseId")
    if (
        isinstance(repository_id, bool)
        or not isinstance(repository_id, int)
        or repository_id <= 0
    ):
        raise SourceRecordRejected(
            "GITHUB_REPOSITORY_ID_MISSING", "linked PR repository ID is missing"
        )
    merge_commit = node.get("mergeCommit")
    merge_oid = merge_commit.get("oid") if isinstance(merge_commit, Mapping) else None
    if (
        node.get("merged") is not True
        or node.get("mergedAt") != pr_metadata.get("merged_at")
        or node.get("baseRefOid") != pr_metadata.get("base_sha")
        or node.get("headRefOid") != pr_metadata.get("head_sha")
        or merge_oid != pr_metadata.get("merge_commit_sha")
    ):
        raise SourceRecordRejected(
            "GITHUB_PR_DETAIL_INCOMPLETE",
            "GraphQL and REST immutable PR fields disagree",
        )
    return number, repository_id

def _capture_pull_request(
    evidence: GitHubEvidence, owner: str, repository: str, number: int
) -> tuple[Mapping[str, Any] | None, dict[str, Any]]:
    root = _api_repo_endpoint(owner, repository)
    payload, entry = evidence.capture(
        method="GET", endpoint=f"{root}/pulls/{number}"
    )
    if not isinstance(payload, Mapping):
        return None, entry
    commit_count = payload.get("commits")
    if (
        isinstance(commit_count, int)
        and not isinstance(commit_count, bool)
        and commit_count > 0
    ):
        for page in range(1, math.ceil(commit_count / 100) + 1):
            evidence.capture(
                method="GET",
                endpoint=f"{root}/pulls/{number}/commits",
                ordered_parameters=(
                    ("per_page", "100"),
                    ("page", str(page)),
                ),
                page=page,
            )
    merge_sha = payload.get("merge_commit_sha")
    head = payload.get("head")
    head_sha = head.get("sha") if isinstance(head, Mapping) else None
    if isinstance(merge_sha, str) and SHA1_PATTERN.fullmatch(merge_sha):
        endpoint_payload, _ = _capture_git_commit(
            evidence, owner, repository, merge_sha
        )
        if isinstance(endpoint_payload, Mapping):
            parents = endpoint_payload.get("parents")
            parent_shas = (
                [item.get("sha") for item in parents if isinstance(item, Mapping)]
                if isinstance(parents, list)
                else []
            )
            snapshot: str | None = None
            if (
                len(parent_shas) == 2
                and isinstance(head_sha, str)
                and parent_shas[1] == head_sha
            ):
                snapshot = parent_shas[0]
            elif len(parent_shas) == 1 and commit_count == 1:
                snapshot = parent_shas[0]
            if isinstance(snapshot, str) and SHA1_PATTERN.fullmatch(snapshot):
                _capture_git_commit(evidence, owner, repository, snapshot)
    return payload, entry


def _read_artifact(root: Path, descriptor: Mapping[str, Any]) -> bytes:
    path = root / str(descriptor.get("path"))
    data = path.read_bytes()
    if (
        sha256_bytes(data) != descriptor.get("sha256")
        or len(data) != descriptor.get("size_bytes")
    ):
        raise DiscoveryExecutionError(
            "RAW_ARTIFACT_HASH_MISMATCH", f"raw artifact mismatch: {path}"
        )
    return data

    """Apply the frozen non-executable grammar to non-bug metadata."""

    if "\x00" in text:
        raise SourceRecordRejected("BUGSINPY_MALFORMED_LINE", "metadata contains NUL")
    values: dict[str, str] = {}
    for line_number, line in enumerate(text.splitlines(), start=1):
        stripped = line.strip()
        if not stripped or stripped.startswith("#"):
            continue
        match = ASSIGNMENT_PATTERN.fullmatch(line)
        if match is None:
            raise SourceRecordRejected(
                "BUGSINPY_MALFORMED_LINE", f"malformed line {line_number}"
            )
        key = match.group("key")
        if key not in allowed:
            raise SourceRecordRejected(
                "BUGSINPY_UNKNOWN_FIELD", f"unknown field {key!r}"
            )
        if key in values:
            raise SourceRecordRejected(
                "BUGSINPY_DUPLICATE_FIELD", f"duplicate field {key!r}"
            )
        raw_value = match.group("value")
        value = raw_value[1:-1] if raw_value[:1] in {"'", '"'} else raw_value
        if not value or any(character in value for character in ("$", "`", "\\")):
            raise SourceRecordRejected(
                "BUGSINPY_UNSAFE_VALUE", f"unsafe value on line {line_number}"
            )
        values[key] = value
    missing = sorted(required - values.keys())
    if missing:
        raise SourceRecordRejected(
            "BUGSINPY_REQUIRED_FIELD_MISSING", "missing fields: " + ", ".join(missing)
        )
    return values

def _capture_issue_relations(
    evidence: GitHubEvidence,
    *,
    owner: str,
    repository: str,
    number: int,
) -> list[Mapping[str, Any]]:
    cursor: str | None = None
    merged_nodes: list[Mapping[str, Any]] = []
    while True:
        variables = {
            "owner": owner,
            "repository": repository,
            "number": number,
            "cursor": cursor,
        }
        payload, _ = evidence.capture(
            method="POST",
            endpoint="https://api.github.com/graphql",
            json_body={"query": GRAPHQL_DOCUMENT, "variables": variables},
            exact_query=GRAPHQL_DOCUMENT,
            graphql_cursor=cursor,
            allow_one_redirect=False,
        )
        if not isinstance(payload, Mapping):
            return []
        if payload.get("errors"):
            raise DiscoveryExecutionError(
                "SOURCE_SCHEMA_DRIFT", "GraphQL response contains errors"
            )
        data = payload.get("data")
        repository_node = data.get("repository") if isinstance(data, Mapping) else None
        issue = (
            repository_node.get("issue")
            if isinstance(repository_node, Mapping)
            else None
        )
        connection = (
            issue.get("closedByPullRequestsReferences")
            if isinstance(issue, Mapping)
            else None
        )
        if not isinstance(connection, Mapping):
            return []
        nodes = connection.get("nodes")
        page_info = connection.get("pageInfo")
        if not isinstance(nodes, list) or not isinstance(page_info, Mapping):
            return []
        if any(not isinstance(node, Mapping) for node in nodes):
            return []
        merged_nodes.extend(node for node in nodes if node.get("merged") is True)
        has_next = page_info.get("hasNextPage")
        end_cursor = page_info.get("endCursor")
        if has_next is False:
            break
        if has_next is not True or not isinstance(end_cursor, str) or not end_cursor:
            return []
        cursor = end_cursor
    return merged_nodes
